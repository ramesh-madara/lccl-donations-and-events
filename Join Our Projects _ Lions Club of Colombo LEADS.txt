<script>
(function () {
    const form = document.getElementById('projectForm');
    if (!form) return;

    const volunteerSection = document.getElementById('volunteerSection');
    const financialSection = document.getElementById('financialSection');
    const organizationSection = document.getElementById('organizationSection');
    const personType = document.getElementById('person_type');

    function updateConditionalSections() {
        const support = Array.from(form.querySelectorAll('input[name="support[]"]:checked')).map(el => el.value);

        const needsVolunteer =
            support.includes('Volunteer My Time') ||
            support.includes('Offer My Professional Skills');

        const needsFinancial =
            support.includes('Make a Financial Donation');

        if (volunteerSection) {
            volunteerSection.classList.toggle('show', needsVolunteer);
        }

        if (financialSection) {
            financialSection.classList.toggle('show', needsFinancial);
        }

        if (organizationSection) {
            organizationSection.classList.toggle(
                'show',
                personType && personType.value === 'Company / Organization'
            );
        }
    }

    form.querySelectorAll('input[name="support[]"]').forEach(function (checkbox) {
        checkbox.addEventListener('change', updateConditionalSections);
    });

    if (personType) {
        personType.addEventListener('change', updateConditionalSections);
    }

    updateConditionalSections();

    form.addEventListener('submit', function (event) {
        const supportChecked = form.querySelectorAll('input[name="support[]"]:checked').length;
        const areasChecked = form.querySelectorAll('input[name="service_areas[]"]:checked').length;

        if (!supportChecked || !areasChecked) {
            event.preventDefault();
            alert('Please select at least one support option and at least one project/service area.');
        }
    });
})();
</script>
